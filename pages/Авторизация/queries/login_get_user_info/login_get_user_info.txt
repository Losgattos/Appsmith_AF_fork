SELECT 
    u.*,
    di.short_name AS ou_short_name,
    di.full_name  AS ou_full_name,
    di.district,
    di.type_id,
    r.system_name,
    r.user_permissions,
    CASE WHEN u.password = {{password.text}} THEN true ELSE false END AS is_valid_password
FROM appsmith.users u
JOIN appsmith.directory_inn di ON u.directory_inn_id = di.directory_inn_id
LEFT JOIN appsmith.user_role r ON u.role_id = r.role_id
WHERE u.login = {{login.text}}
LIMIT 1;
