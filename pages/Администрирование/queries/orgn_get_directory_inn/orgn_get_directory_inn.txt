SELECT directory_inn_id, inn, kpp, ogrn, short_name, full_name, print_name, tip_ou, status_ou, opf, parent_short_name, parent_print_name, lsfusion_key
FROM appsmith.directory_inn
WHERE ( inn ILIKE '%{{ organisationTable.searchText || "" }}%'
     OR short_name ILIKE '%{{ organisationTable.searchText || "" }}%'
     OR full_name ILIKE '%{{ organisationTable.searchText || "" }}%' )
  AND ( '{{ appsmith.store.orgn_iogv_filter || "" }}' = ''
     OR iogv_inn = '{{ appsmith.store.orgn_iogv_filter }}' )
ORDER BY "{{ organisationTable.sortOrder.column || 'short_name' }}" {{ organisationTable.sortOrder.order || 'ASC' }}
LIMIT {{ organisationTable.pageSize }}
OFFSET {{ (organisationTable.pageNo - 1) * organisationTable.pageSize }}
